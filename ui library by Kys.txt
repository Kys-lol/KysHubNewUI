local function log(...) print(PREFIX, ...) end
local function warnx(...) warn(PREFIX, ...) end

local function tween(object, info, properties)
    if not object then return end
    local t = TweenService:Create(object, info, properties)
    t:Play()
    return t
end

local function safeCall(fn, ...)
    if type(fn) == "function" then
        local ok, err = pcall(fn, ...)
        if not ok then warnx("Callback Exception:", err) end
    end
end

local function getRobloxGui() return CoreGui:FindFirstChild("RobloxGui") end

local function findRecursive(parent, name, className)
    if not parent then return nil end
    local direct = parent:FindFirstChild(name)
    if direct and (not className or direct:IsA(className)) then return direct end
    for _, obj in ipairs(parent:GetDescendants()) do
        if obj.Name == name and (not className or obj:IsA(className)) then return obj end
    end
    return nil
end

local function findHelpContainer()
    local RobloxGui = getRobloxGui()
    if not RobloxGui then return nil end
    return findRecursive(RobloxGui, "HelpPageContainer", "Frame")
        or findRecursive(RobloxGui, "HelpPage", "Frame")
        or findRecursive(RobloxGui, "HelpTabContainer", "Frame")
end

local function findHelpTab()
    local RobloxGui = getRobloxGui()
    if not RobloxGui then return nil end
    local hubBar = findRecursive(RobloxGui, "HubBarContainer")
    if not hubBar then return nil end
    return hubBar:FindFirstChild("HelpTab") or findRecursive(hubBar, "HelpTab", "GuiObject")
end

local function corner(object, radius)
    local c = object:FindFirstChild("KysCorner") or Instance.new("UICorner")
    c.Name = "KysCorner"
    c.CornerRadius = UDim.new(0, radius or 8)
    c.Parent = object
    return c
end

local function stroke(object, color, trans)
    local s = object:FindFirstChild("KysStroke") or Instance.new("UIStroke")
    s.Name = "KysStroke"
    s.Thickness = 1
    s.Color = color or Theme.Border
    s.Transparency = trans or 0.4
    s.Parent = object
    return s
end

local function createText(parent, name, text, size, color, font)
    local label = Instance.new("TextLabel")
    label.Name = name
    label.BackgroundTransparency = 1
    label.Text = text or ""
    label.TextSize = size or 14
    label.TextColor3 = color or Theme.Text
    label.Font = font or Enum.Font.Gotham
    label.TextXAlignment = Enum.TextXAlignment.Left
    label.TextYAlignment = Enum.TextYAlignment.Center
    label.Parent = parent
    return label
end

local function createIcon(parent, iconKey, size, color)
    local container = Instance.new("Frame")
    container.Name = "IconContainer_" .. tostring(iconKey)
    container.BackgroundTransparency = 1
    container.Size = UDim2.fromOffset(size or 18, size or 18)
    container.Parent = parent

    local info = IconMap[iconKey] or IconMap["alert-triangle"]
    
    if info.Asset and info.Asset ~= "" then
        local img = Instance.new("ImageLabel")
        img.Name = "IconImage"
        img.BackgroundTransparency = 1
        img.Size = UDim2.new(1, 0, 1, 0)
        img.Image = info.Asset
        if color then img.ImageColor3 = color end
        img.Parent = container
    else
        local txt = createText(container, "IconText", info.Fallback, size or 14, color or Theme.Text, Enum.Font.GothamBold)
        txt.TextXAlignment = Enum.TextXAlignment.Center
        txt.Size = UDim2.new(1, 0, 1, 0)
    end

    return container
end

local function disableParentsClipping(object)
    local current = object
    while current and current ~= game and not current:IsA("ScreenGui") do
        if current:IsA("GuiObject") then current.ClipsDescendants = false end
        current = current.Parent
    end
end

-- ============================================================
-- TAB CLASS & EXTENDED APIS

local TabClass = {}
TabClass.__index = TabClass

function TabClass:_addItem(item)
    table.insert(self._items, item)
    if self._library.ActiveTab == self then self._library:_updateContent() end
    return item
end

function TabClass:AddButton(config)
    config = config or {}
    return self:_addItem({
        Type = "Button",
        Name = config.Name or "Button",
        Description = config.Description or "",
        Icon = config.Icon or "chevron-right",
        Callback = config.Callback,
        Order = config.Order or (#self._items + 1),
    })
end

function TabClass:AddToggle(config)
    config = config or {}
    local flag = config.Flag or (self.Name .. "_" .. (config.Name or "Toggle"))
    if RuntimeState[flag] == nil then RuntimeState[flag] = config.CurrentValue == true end
    return self:_addItem({
        Type = "Toggle",
        Name = config.Name or "Toggle",
        Description = config.Description or "",
        Icon = config.Icon or "check",
        Flag = flag,
        Callback = config.Callback,
        Order = config.Order or (#self._items + 1),
    })
end

function TabClass:AddInfo(config)
    config = config or {}
    return self:_addItem({
        Type = "Info",
        Title = config.Title or "Information",
        Message = config.Message or "",
        Order = config.Order or (#self._items + 1),
    })
end

function TabClass:AddDashboard(config)
    config = config or {}
    return self:_addItem({
        Type = "Dashboard",
        Title = config.Title or "System Dashboard",
        Stats = config.Stats or { { Title = "Status", Value = "Active" } },
        Logs = config.Logs or { "System initialized." },
        Order = config.Order or (#self._items + 1),
    })
end

function TabClass:AddTextInput(config)
    config = config or {}
    return self:_addItem({
        Type = "TextInput",
        Name = config.Name or "Input",
        Placeholder = config.Placeholder or "Enter text...",
        Callback = config.Callback,
        Order = config.Order or (#self._items + 1),
    })
end

local Library = {}
Library.__index = Library

function Library:SetTheme(newTheme)
    for k, v in pairs(newTheme) do
        Theme[k] = v
    end
    if self.Root then self:_buildUI() end
end

function Library:SetMenuName(name)
    self.MenuName = name
    self:_forceMenuName()
end

function Library:SetStudioTitle(title)
    self.StudioTitle = title
    if self.SidebarTitleLabel then
        self.SidebarTitleLabel.Text = string.upper(title)
    end
end

function Library:AddTab(config)
    config = config or {}
    local tab = setmetatable({
        Name = config.Name or "Tab",
        Icon = config.Icon or "settings",
        Order = config.Order or (#self.Tabs + 1),
        _items = {},
        _library = self,
        _emptyMessage = config.EmptyMessage or "No features available in this tab.",
    }, TabClass)
    table.insert(self.Tabs, tab)
    if not self.ActiveTab then self.ActiveTab = tab end
    if self.Root then self:_buildUI() end
    return tab
end

-- API SYSTEM SCAN & ACCELERATION ROBLOX UI
function Library:SetUISpeed(speedMultiplier)
    speedMultiplier = math.clamp(speedMultiplier or 1, 0.1, 10)
    local robloxGui = getRobloxGui()
    if not robloxGui then return end

    local count = 0
    for _, obj in ipairs(robloxGui:GetDescendants()) do
        if obj:IsA("Tween") then
            obj:Cancel()
            count = count + 1
        end
    end
    
    log("UI Scan & Speed adjustment applied. Processed elements:", count, "Multiplier:", speedMultiplier)
    return count
end

function Library:_forceMenuName()
    local helpTab = findHelpTab()
    if not helpTab then return end
    local tabLabel = findRecursive(helpTab, "TabLabel", "GuiObject")
    if not tabLabel then return end
    local title = findRecursive(tabLabel, "Title")
    if title and (title:IsA("TextLabel") or title:IsA("TextButton") or title:IsA("TextBox")) then
        title.Text = self.MenuName or "Library Studio"
    end
end

    local root = Instance.new("Frame")
    root.Name = "KysMenuContent"

function Library:_updateRootBounds()
    if not self.Root or not self.Root.Parent then return end
    local parent = self.Root.Parent
    disableParentsClipping(parent)

    local pageView = parent
    while pageView and pageView.Name ~= "PageView" and pageView.Name ~= "Page" and pageView.Parent ~= game do
        pageView = pageView.Parent
    end

    if pageView and pageView:IsA("GuiObject") then
        local targetSize = pageView.AbsoluteSize
        local targetPos = pageView.AbsolutePosition
        local parentPos = parent.AbsolutePosition
        self.Root.Position = UDim2.fromOffset(targetPos.X - parentPos.X, targetPos.Y - parentPos.Y)
        self.Root.Size = UDim2.fromOffset(targetSize.X, targetSize.Y)
    else
        self.Root.Position = UDim2.fromOffset(-15, -60)
        self.Root.Size = UDim2.new(1, 30, 1, 120)
    end
end

function Library:_updateTabStyles()
    local tweenInfo = TweenInfo.new(0.25, Enum.EasingStyle.Quad, Enum.EasingDirection.Out)
    for tab, elements in pairs(self.TabButtons) do
        local selected = self.ActiveTab == tab
        local targetColor = selected and Theme.SurfaceActive or Theme.Surface
        local targetTrans = selected and 0.2 or 1
        local targetTextCol = selected and Theme.White or Theme.SubText

        tween(elements.Button, tweenInfo, {
            BackgroundColor3 = targetColor,
            BackgroundTransparency = targetTrans
        })
        tween(elements.Label, tweenInfo, { TextColor3 = targetTextCol })
        
        for _, child in ipairs(elements.Icon:GetChildren()) do
            if child:IsA("ImageLabel") then tween(child, tweenInfo, { ImageColor3 = targetTextCol }) end
            if child:IsA("TextLabel") then tween(child, tweenInfo, { TextColor3 = targetTextCol }) end
        end
    end
end

function Library:_updateContent()
    if not self.ContentHeader or not self.FeatureArea then return end
    
    self.ContentHeader.Text = self.ActiveTab and self.ActiveTab.Name or "Menu"
    self.FeatureArea:ClearAllChildren()

    local layout = Instance.new("UIListLayout")
    layout.SortOrder = Enum.SortOrder.LayoutOrder
    layout.Padding = UDim.new(0, 8)
    layout.Parent = self.FeatureArea

    self.FeatureArea.Position = UDim2.fromOffset(0, 72)
    tween(self.FeatureArea, TweenInfo.new(0.3, Enum.EasingStyle.Quad, Enum.EasingDirection.Out), {
        Position = UDim2.fromOffset(0, 62)
    })

    if not self.ActiveTab or #self.ActiveTab._items == 0 then
        local empty = Instance.new("Frame")
        empty.Name = "EmptyState"
        empty.BackgroundTransparency = 1
        empty.AnchorPoint = Vector2.new(0.5, 0.5)
        empty.Position = UDim2.new(0.5, 0, 0.4, 0)
        empty.Size = UDim2.new(1, -20, 0, 100)
        empty.ZIndex = 1010
        empty.Parent = self.FeatureArea

        local errIcon = createIcon(empty, "alert-triangle", 44, Theme.Error)
        errIcon.AnchorPoint = Vector2.new(0.5, 0)
        errIcon.Position = UDim2.new(0.5, 0, 0, 0)
        errIcon.ZIndex = 1011

        local errorText = createText(empty, "ErrorText", self.ActiveTab and self.ActiveTab._emptyMessage or "No components available.", 14, Theme.Text, Enum.Font.GothamMedium)
        errorText.TextXAlignment = Enum.TextXAlignment.Center
        errorText.Position = UDim2.fromOffset(0, 52)
        errorText.Size = UDim2.new(1, 0, 0, 24)
        return
    end

    for _, item in ipairs(self.ActiveTab._items) do
        if item.Type == "Button" then
            self:_renderButton(self.FeatureArea, item)
        elseif item.Type == "Toggle" then
            self:_renderToggle(self.FeatureArea, item)
        elseif item.Type == "Info" then
            self:_renderInfo(self.FeatureArea, item)
        elseif item.Type == "Dashboard" then
            self:_renderDashboard(self.FeatureArea, item)
        elseif item.Type == "TextInput" then
            self:_renderTextInput(self.FeatureArea, item)
        end
    end
end

function Library:_buildUI()
    if not self.Root or not self.Root.Parent then return end
    self.Root:ClearAllChildren()

    self:_updateRootBounds()

    local width = self.Root.AbsoluteSize.X
    local margin = 12
    local sidebarWidth = math.clamp(math.floor(width * 0.22), 140, 200)
    local gap = 16
    local contentX = margin + sidebarWidth + gap

    self:_createSidebar(self.Root, sidebarWidth, margin)
    self:_createContentArea(self.Root, contentX, margin)
end

function Library:Mount(container)
    if not container then return false end

    for _, child in ipairs(container:GetChildren()) do
        if child:IsA("GuiObject") and child.Name ~= "KysMenuContent" then
            child.Visible = false
        end
    end

    local root = container:FindFirstChild("KysMenuContent")
    if root and not root:IsA("Frame") then
        root:Destroy()
        root = nil
    end

    if not root then root = self:_createRoot(container) end
    root.Visible = true

    self.Root = root
    self:_buildUI()

    if self._sizeConnection then self._sizeConnection:Disconnect() end
    self._sizeConnection = container:GetPropertyChangedSignal("AbsoluteSize"):Connect(function()
        self:_updateRootBounds()
    end)

    return true
end

local function CreateLibrary()
    return setmetatable({
        Tabs = {},
        ActiveTab = nil,
        Root = nil,
        MenuName = "Library Studio",
        StudioTitle = "Studio Hub",
        _sizeConnection = nil,
    }, Library)
end

local KysMenu = CreateLibrary()

local function ensureMenu()
    local container = findHelpContainer()
    if not container then return false end

    for _, child in ipairs(container:GetChildren()) do
        if child:IsA("GuiObject") and child.Name ~= "KysMenuContent" then
            child.Visible = false
        end
    end

    if not KysMenu.Root or not KysMenu.Root.Parent then
        KysMenu:Mount(container)
    else
        KysMenu.Root.Visible = true
    end

    KysMenu:_forceMenuName()
    return true
end

task.spawn(function()
    log("Kys UI Library starting...")
    task.wait(1)

    for attempt = 1, INITIAL_RETRY do
        local ok, result = pcall(ensureMenu)
        if ok and result then break end
        task.wait(0.5)
    end

    while task.wait(CHECK_INTERVAL) do
        pcall(ensureMenu)
    end
end)

return KysMenu
